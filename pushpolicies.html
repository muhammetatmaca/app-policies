    id                INTEGER PRIMARY KEY AUTOINCREMENT,
    package_name      TEXT NOT NULL,         -- E.g., 'com.whatsapp'

    message_body      TEXT,                  -- Notification preview content

    is_retracted      INTEGER DEFAULT 0,     -- 1 if sender deleted message later
    priority_lane     TEXT DEFAULT 'NORMAL', -- 'VIP', 'NORMAL', 'MUTED'
    checksum_sha256   TEXT NOT NULL          -- Cryptographic payload integrity

    total_alerts      INTEGER DEFAULT 0,
    vip_status        INTEGER DEFAULT 0,
    last_active_at    INTEGER NOT NULL
);
-- -------------------------------------------------------------------------
-- TABLE: retention_policies // Automated cleanup rules
